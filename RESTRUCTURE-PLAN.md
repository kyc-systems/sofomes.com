## 🔧 FUNCIONALIDADES TÉCNICAS

### 1. Buscador SOFOMes (JavaScript)

**Archivo:** `assets/js/directory.js`

```javascript
class SOFOMDirectory {
  constructor(jsonPath) {
    this.data = null;
    this.filteredData = null;
    this.loadData(jsonPath);
  }

  async loadData(path) {
    const response = await fetch(path);
    const json = await response.json();
    this.data = json.sofomes;
    this.filteredData = this.data;
    this.render();
  }

  filter(query, state, type) {
    this.filteredData = this.data.filter(sofom => {
      const matchQuery = query === '' ||
        sofom.denominacion_social.toLowerCase().includes(query.toLowerCase()) ||
        sofom.nombre_corto.toLowerCase().includes(query.toLowerCase());

      const matchState = state === 'all' || sofom.estado === state;
      const matchType = type === 'all' ||
        (type === 'er' && sofom.sector.includes('E.R.')) ||
        (type === 'enr' && sofom.sector.includes('E.N.R.'));

      return matchQuery && matchState && matchType;
    });

    this.render();
  }

  render() {
    // Renderizar tabla + stats
    document.getElementById('total-results').textContent = this.filteredData.length;
    // ... renderizado tabla
  }

  exportCSV() {
    // Exportar filteredData a CSV
  }
}
```
