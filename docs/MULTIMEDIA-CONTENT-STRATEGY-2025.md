## 4. OPTIMIZACIÓN DE IMÁGENES (AVIF/WebP)

### Pipeline de Optimización Automática

```javascript
// optimize-images-pipeline.js
const sharp = require('sharp');
const glob = require('glob');
const path = require('path');

const SIZES = {
  hero: [640, 1280, 1920],
  card: [400, 800],
  thumbnail: [300, 600],
  full: [1280, 1920, 2560]
};

const FORMATS = ['avif', 'webp', 'jpeg'];

async function optimizePipeline() {
  const images = glob.sync('./assets/images/raw/**/*.{jpg,png}');

  for (const imagePath of images) {
    const filename = path.basename(imagePath, path.extname(imagePath));
    const category = path.dirname(imagePath).split('/').pop();
    const sizes = SIZES[category] || SIZES.full;

    console.log(`Processing: ${filename} (${category})`);

    for (const size of sizes) {
      const pipeline = sharp(imagePath)
        .resize(size, null, {
          withoutEnlargement: true,
          kernel: sharp.kernel.lanczos3
        })
        .rotate(); // Auto-rotate based on EXIF

      // AVIF (best compression 2025)
      await pipeline.clone()
        .avif({
          quality: category === 'hero' ? 80 : 75,
          effort: 6,
          chromaSubsampling: '4:2:0'
        })
        .toFile(`./dist/images/${category}/${filename}-${size}.avif`);

      // WebP (fallback 1)
      await pipeline.clone()
        .webp({
          quality: category === 'hero' ? 85 : 80,
          effort: 6,
          smartSubsample: true
        })
        .toFile(`./dist/images/${category}/${filename}-${size}.webp`);

      // JPEG (fallback 2)
      await pipeline.clone()
        .jpeg({
          quality: category === 'hero' ? 90 : 85,
          progressive: true,
          mozjpeg: true
        })
        .toFile(`./dist/images/${category}/${filename}-${size}.jpg`);

      console.log(`  ✓ Generated ${filename}-${size} in 3 formats`);
    }
  }

  console.log('✅ Image optimization complete');
}

optimizePipeline();
```

```javascript
// multimedia-tracking.js

// Track video play
document.querySelectorAll('video').forEach(video => {
  video.addEventListener('play', () => {
    gtag('event', 'video_play', {
      event_category: 'Video Engagement',
      event_label: video.dataset.videoId,
      value: 1
    });
  });

  // Track milestones (25%, 50%, 75%, 100%)
  let milestones = [0.25, 0.5, 0.75, 1];
  video.addEventListener('timeupdate', () => {
    const progress = video.currentTime / video.duration;

    milestones = milestones.filter(milestone => {
      if (progress >= milestone) {
        gtag('event', 'video_progress', {
          event_category: 'Video Engagement',
          event_label: video.dataset.videoId,
          value: Math.round(milestone * 100)
        });
        return false;
      }
      return true;
    });
  });
});

// Track infografía interactions
document.querySelectorAll('.infografia-interactiva').forEach(infografia => {
  infografia.addEventListener('click', (e) => {
    if (e.target.closest('[data-info]')) {
      gtag('event', 'infografia_interaction', {
        event_category: 'Infografía Engagement',
        event_label: infografia.dataset.infografiaId,
        action: e.target.dataset.info
      });
    }
  });

  // Track hover time
  let hoverStart = null;
  infografia.addEventListener('mouseenter', () => {
    hoverStart = Date.now();
  });

  infografia.addEventListener('mouseleave', () => {
    if (hoverStart) {
      const hoverTime = Date.now() - hoverStart;
      gtag('event', 'infografia_time', {
        event_category: 'Infografía Engagement',
        event_label: infografia.dataset.infografiaId,
        value: Math.round(hoverTime / 1000) // seconds
      });
    }
  });
});
```
